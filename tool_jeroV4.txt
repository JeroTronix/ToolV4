class Color:
    activo = True
    tema = "noche"
    _borde = "\033[38;5;67m"
    _valor = "\033[38;5;252m"
    _etiqueta = "\033[38;5;109m"
    _ok = "\033[38;5;71m"
    _aviso = "\033[38;5;179m"
    _malo = "\033[38;5;167m"
    _info = "\033[38;5;103m"
    _tenue = "\033[38;5;240m"
    _titulo = "\033[1;38;5;110m"

    PALETAS = {
        "noche": ("67", "252", "109", "71", "179", "167", "103", "240", "110"),
        "hielo": ("45", "253", "117", "80", "222", "204", "152", "243", "81"),
        "matrix": ("28", "120", "40", "46", "178", "160", "34", "22", "40"),
        "ambar": ("136", "223", "179", "107", "214", "166", "138", "94", "179"),
        "mono": ("240", "255", "245", "255", "255", "255", "245", "240", "255"),
    }

    @classmethod
    def aplicar(cls, nombre):
        p = cls.PALETAS.get(str(nombre).lower())
        if not p:
            return False
        cls.tema = str(nombre).lower()
        (cls._borde, cls._valor, cls._etiqueta, cls._ok, cls._aviso,
         cls._malo, cls._info, cls._tenue, cls._titulo) = ["\033[38;5;%sm" % c for c in p]
        return True

    @classmethod
    def _w(cls, cod, txt):
        if not cls.activo or os.environ.get("NO_COLOR"):
            return str(txt)
        return cod + str(txt) + "\033[0m"

    @classmethod
    def borde(cls, t): return cls._w(cls._borde, t)
    @classmethod
    def valor(cls, t): return cls._w(cls._valor, t)
    @classmethod
    def etiqueta(cls, t): return cls._w(cls._etiqueta, t)
    @classmethod
    def ok(cls, t): return cls._w(cls._ok, t)
    @classmethod
    def aviso(cls, t): return cls._w(cls._aviso, t)
    @classmethod
    def malo(cls, t): return cls._w(cls._malo, t)
    @classmethod
    def info(cls, t): return cls._w(cls._info, t)
    @classmethod
    def tenue(cls, t): return cls._w(cls._tenue, t)
    @classmethod
    def titulo(cls, t): return cls._w(cls._titulo, t)


def ancho(texto):
    limpio = re.sub(r"\033\[[0-9;]*m", "", str(texto))
    return len(limpio)


def pad(texto, n):
    return str(texto) + " " * max(0, n - ancho(texto))

class Cripto:
    """Fernet si hay cryptography; si no, XOR+HMAC con la misma clave."""

    @staticmethod
    def _clave():
        os.makedirs(DIR_BASE, exist_ok=True)
        if not os.path.exists(RUTA_CLAVE):
            semilla = os.urandom(32)
            with open(RUTA_CLAVE, "wb") as f:
                f.write(semilla)
            try:
                os.chmod(RUTA_CLAVE, 0o600)
            except Exception:
                pass
        with open(RUTA_CLAVE, "rb") as f:
            return f.read()

    @classmethod
    def cifrar(cls, texto):
        if texto in (None, ""):
            return ""
        bruto = str(texto).encode()
        semilla = cls._clave()
        if HAY_CRIPTO:
            try:
                return "f:" + Fernet(base64.urlsafe_b64encode(semilla[:32])).encrypt(bruto).decode()
            except Exception:
                pass
        flujo = (semilla * (len(bruto) // 32 + 1))[:len(bruto)]
        cuerpo = bytes(a ^ b for a, b in zip(bruto, flujo))
        mac = hmac.new(semilla, cuerpo, hashlib.sha256).hexdigest()[:32]
        return "x:%s:%s" % (base64.b64encode(cuerpo).decode(), mac)

    @classmethod
    def descifrar(cls, dato):
        if not dato:
            return ""
        dato = str(dato)
        semilla = cls._clave()
        if dato.startswith("f:") and HAY_CRIPTO:
            try:
                return Fernet(base64.urlsafe_b64encode(semilla[:32])).decrypt(
                    dato[2:].encode()).decode()
            except (InvalidToken, Exception):
                return ""
        if dato.startswith("x:"):
            try:
                _, b64, mac = dato.split(":", 2)
                cuerpo = base64.b64decode(b64)
                if hmac.new(semilla, cuerpo, hashlib.sha256).hexdigest()[:32] != mac:
                    return ""
                flujo = (semilla * (len(cuerpo) // 32 + 1))[:len(cuerpo)]
                return bytes(a ^ b for a, b in zip(cuerpo, flujo)).decode()
            except Exception:
                return ""
        return dato

class Config:
    def __init__(self):
        self.datos = dict(POR_DEFECTO)
        self._lock = threading.Lock()
        self.cargar()

    def cargar(self):
        if not os.path.exists(RUTA_CONFIG):
            return
        try:
            with open(RUTA_CONFIG, "r", encoding="utf-8") as f:
                guardado = json.load(f)
        except Exception:
            return
        for k, v in guardado.items():
            if k in POR_DEFECTO:
                if k in CLAVES_SECRETAS and v:
                    self.datos[k] = Cripto.descifrar(v) or v
                else:
                    self.datos[k] = v
        if not Color.aplicar(self.datos.get("tema", "noche")):
            Color.aplicar("noche")

    def guardar(self):
        with self._lock:
            os.makedirs(DIR_BASE, exist_ok=True)
            salida = {}
            for k, v in self.datos.items():
                if k in CLAVES_SECRETAS and v:
                    salida[k] = Cripto.cifrar(v)
                else:
                    salida[k] = v
            temporal = RUTA_CONFIG + ".tmp"
            with open(temporal, "w", encoding="utf-8") as f:
                json.dump(salida, f, ensure_ascii=False, indent=2)
            os.replace(temporal, RUTA_CONFIG)
            try:
                os.chmod(RUTA_CONFIG, 0o600)
            except Exception:
                pass

    def get(self, clave, defecto=None):
        return self.datos.get(clave, POR_DEFECTO.get(clave, defecto))

    def set(self, clave, valor):
        if clave not in POR_DEFECTO:
            return False, "clave desconocida"
        tipo = type(POR_DEFECTO[clave])
        try:
            if tipo is bool:
                if str(valor).lower() in ("1", "si", "sí", "true", "on", "yes"):
                    valor = True
                elif str(valor).lower() in ("0", "no", "false", "off"):
                    valor = False
                else:
                    return False, "usa si/no"
            elif tipo is int:
                valor = int(valor)
            else:
                valor = str(valor)
        except Exception:
            return False, "valor invalido"
        self.datos[clave] = valor
        if clave == "tema":
            Color.aplicar(valor)
        return True, "ok"

def ahora():
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")


def marca_tiempo():
    return datetime.now().strftime("%Y%m%d_%H%M%S")


def es_ip(t):
    try:
        socket.inet_aton(str(t))
        return str(t).count(".") == 3
    except Exception:
        return False


def es_ipv6(t):
    try:
        socket.inet_pton(socket.AF_INET6, str(t))
        return True
    except Exception:
        return False


def resolver(host):
    if es_ip(host):
        return str(host)
    try:
        return socket.gethostbyname(str(host))
    except Exception:
        return None


def es_privada(ip):
    p = str(ip).split(".")
    if len(p) != 4:
        return False
    try:
        a, b = int(p[0]), int(p[1])
    except ValueError:
        return False
    return (a in (10, 127) or (a == 192 and b == 168)
            or (a == 172 and 16 <= b <= 31) or (a == 169 and b == 254))


def dominio_limpio(t):
    t = str(t).strip().lower()
    t = re.sub(r"^https?://", "", t)
    t = t.split("/")[0].split("?")[0]
    return t


def correo_valido(t):
    return bool(re.fullmatch(r"[^@\s]+@[^@\s]+\.[A-Za-z]{2,}", str(t).strip()))


def ua():
    return random.choice(UA_POOL)


def paralelo(tareas, limite=HILOS):
    salida = {k: None for k in tareas}
    if not tareas:
        return salida
    with ThreadPoolExecutor(max_workers=min(limite, len(tareas))) as ex:
        futuros = {n: ex.submit(f) for n, f in tareas.items()}
        for n, f in futuros.items():
            try:
                salida[n] = f.result(timeout=35)
            except Exception:
                salida[n] = None
    return salida

class Red:
    def __init__(self, config):
        self.config = config
        self._cache = {}
        self._lock = threading.Lock()

    def _cab(self, extra=None):
        cab = {"User-Agent": ua(), "Accept": "*/*",
               "Accept-Language": "es,en;q=0.8",
               "Accept-Encoding": "identity", "Connection": "close"}
        if extra:
            cab.update(extra)
        return cab

    def get(self, url, timeout=None, headers=None, datos=None, metodo="GET",
            reintentos=None, cache=True):
        if cache:
            with self._lock:
                if url in self._cache:
                    return self._cache[url]
        intentos = max(1, int(reintentos if reintentos is not None
                              else self.config.get("reintentos", 2)))
        ultimo = (None, "")
        for i in range(intentos):
            try:
                req = urllib.request.Request(url, headers=self._cab(headers),
                                             data=datos, method=metodo)
                with urllib.request.urlopen(
                        req, timeout=timeout or self.config.get("timeout", 9)) as r:
                    cuerpo = r.read(LARGO_MAX).decode("utf-8", "replace")
                    res = (r.getcode(), cuerpo)
                    break
            except urllib.error.HTTPError as e:
                try:
                    cuerpo = e.read(200000).decode("utf-8", "replace")
                except Exception:
                    cuerpo = ""
                if e.code not in (429, 500, 502, 503, 504):
                    res = (e.code, cuerpo)
                    break
                ultimo = (e.code, cuerpo)
            except Exception as e:
                ultimo = (None, str(e))
            time.sleep(0.35 * (i + 1) + random.random() * 0.3)
        else:
            res = ultimo
        if cache and res[0] == 200:
            with self._lock:
                self._cache[url] = res
        return res

    def json(self, url, **kw):
        code, cuerpo = self.get(url, **kw)
        if not cuerpo:
            return None
        try:
            return json.loads(cuerpo)
        except Exception:
            return None

    def limpiar(self):
        with self._lock:
            self._cache.clear()

class UI:
    def __init__(self, config):
        self.config = config
        self.silencio = False

    def _p(self, marca, texto):
        if self.silencio:
            return
        print(marca + " " + str(texto))

    def crudo(self, texto=""):
        if not self.silencio:
            print(texto)

    def escribir(self, texto=""):
        self.crudo(texto)

    def sep(self, caracter="─", largo=62):
        self.crudo(Color.tenue(caracter * largo))

    def titulo(self, texto, estilo=None):
        self.crudo()
        self.crudo(Color.titulo("┌─ " + str(texto)))
        self.crudo(Color.borde("└" + "─" * min(len(str(texto)) + 3, 62)))

    def sub(self, texto):
        self.crudo("  " + Color.etiqueta("▪ " + str(texto)))

    def campo(self, etiqueta, valor):
        if valor in (None, "", [], {}):
            return
        self.crudo("  " + Color.etiqueta(pad(str(etiqueta), 18) + "│ ")
                   + Color.valor(valor))

    def lista(self, etiqueta, valores, tope=15):
        valores = [v for v in (valores or []) if v]
        if not valores:
            return
        self.sub(etiqueta)
        for v in valores[:tope]:
            self.crudo("  " + Color.borde("├─ ") + Color.valor(v))

    def enlace(self, etiqueta, url):
        self.crudo("  " + Color.etiqueta(pad(str(etiqueta), 16)) + Color.info(url))

    def ok(self, texto): self._p(Color.ok("[+]"), texto)
    def malo(self, texto): self._p(Color.malo("[-]"), texto)
    def aviso(self, texto): self._p(Color.aviso("[!]"), texto)
    def info(self, texto): self._p(Color.info("[*]"), texto)

    def mapa(self, lat, lon):
        if lat and lon:
            self.crudo("  " + Color.etiqueta(pad("mapa", 18) + "│ ")
                       + Color.info("https://www.google.com/maps?q=%s,%s" % (lat, lon)))

class Anim:
    MARCOS = {
        "radar": "▁▂▃▄▅▆▇█▇▆▅▄▃▂",
        "ondas": "·∙•●•∙·",
        "reticula": "◜◝◞◟",
        "sobre": "▱▰▱▰",
        "arbol": "▹▸▹▸",
        "remolino": "◐◓◑◒",
    }

    def __init__(self, config, ui):
        self.config = config
        self.ui = ui
        self.activo = bool(config.get("animacion", True)) and sys.stdout.isatty()

    def esperar(self, etiqueta, fn, estilo="radar", minimo=0.25):
        if not self.activo or self.ui.silencio:
            self.ui.info(etiqueta)
            return fn()
        marcos = self.MARCOS.get(estilo, self.MARCOS["radar"])
        fin = threading.Event()
        resultado = {}

        def obrero():
            try:
                resultado["v"] = fn()
            except Exception as e:
                resultado["e"] = e
            finally:
                fin.set()

        h = threading.Thread(target=obrero, daemon=True)
        inicio = time.time()
        h.start()
        i = 0
        while not fin.is_set():
            self.ui.crudo("\r  " + Color.info(marcos[i % len(marcos)] + " ")
                          + Color.etiqueta(etiqueta) + "   ")
            i += 1
            time.sleep(0.08)
        while time.time() - inicio < minimo:
            time.sleep(0.02)
        self.ui.crudo("\r" + " " * (ancho(etiqueta) + 8) + "\r")
        if "e" in resultado:
            self.ui.malo("%s: %s" % (etiqueta, resultado["e"]))
            return None
        return resultado.get("v")

class Contexto:
    def __init__(self, config, ui, red, anim, flags):
        self.config = config
        self.ui = ui
        self.red = red
        self.anim = anim
        self.flags = flags

    def archivar(self, tipo, objetivo, datos):
        if not self.config.get("archivar", True) or self.flags.get("no_archivo"):
            return
        try:
            os.makedirs(RUTA_DOSSIERS, exist_ok=True)
            nombre = "%s_%s_%s.json" % (
                marca_tiempo(), tipo, re.sub(r"\W+", "_", str(objetivo))[:50])
            ruta = os.path.join(RUTA_DOSSIERS, nombre)
            with open(ruta, "w", encoding="utf-8") as f:
                json.dump({"tipo": tipo, "objetivo": str(objetivo),
                           "fecha": ahora(), "datos": datos},
                          f, ensure_ascii=False, indent=2)
            self.ui.ok("dossier -> " + ruta)
        except Exception as e:
            self.ui.aviso("no se pudo archivar: %s" % e)

    def emitir(self, datos):
        if self.flags.get("json") or self.config.get("json"):
            print(json.dumps(datos, ensure_ascii=False, indent=2))

def _geo(ctx, ip):
    q = urllib.parse.quote(ip)
    d = {}
    res = paralelo({
        "ipwho": lambda: ctx.red.json("https://ipwho.is/" + q),
        "ipapiis": lambda: ctx.red.json("https://api.ipapi.is/?q=" + q),
        "ipapi": lambda: ctx.red.json(
            "http://ip-api.com/json/%s?fields=status,country,countryCode,"
            "regionName,city,zip,lat,lon,timezone,isp,org,as,reverse,mobile,"
            "proxy,hosting" % q),
        "ipinfo": lambda: ctx.red.json("https://ipinfo.io/%s/json" % q),
        "freeip": lambda: ctx.red.json("https://freeipapi.com/api/json/" + q),
        "iploc": lambda: ctx.red.json("https://api.iplocation.net/?ip=" + q),
        "hostip": lambda: ctx.red.json("https://api.hostip.info/get_json.php?ip=" + q),
    })

    def pon(k, v):
        if v not in (None, "", 0, "-") and not d.get(k):
            d[k] = v

    a = res.get("ipwho") or {}
    if a.get("success") is not False:
        pon("pais", a.get("country"))
        pon("codigo", a.get("country_code"))
        pon("region", a.get("region"))
        pon("ciudad", a.get("city"))
        pon("cp", a.get("postal"))
        pon("lat", a.get("latitude"))
        pon("lon", a.get("longitude"))
        pon("zona", (a.get("timezone") or {}).get("id"))
        con = a.get("connection") or {}
        pon("isp", con.get("isp"))
        pon("org", con.get("org"))
        if con.get("asn"):
            pon("asn", "AS%s" % con["asn"])

    b = res.get("ipapiis") or {}
    pon("vpn", b.get("is_vpn"))
    pon("proxy", b.get("is_proxy"))
    pon("tor", b.get("is_tor"))
    pon("hosting", b.get("is_datacenter"))
    pon("abuser", b.get("is_abuser"))
    pon("crawler", b.get("is_crawler"))
    pon("empresa", (b.get("company") or {}).get("name"))
    pon("org", (b.get("asn") or {}).get("org"))
    pon("asn", (b.get("asn") or {}).get("asn"))
    loc = b.get("location") or {}
    pon("pais", loc.get("country"))
    pon("ciudad", loc.get("city"))
    pon("region", loc.get("state"))
    pon("lat", loc.get("latitude"))
    pon("lon", loc.get("longitude"))
    pon("zona", loc.get("timezone"))

    e = res.get("ipapi") or {}
    if e.get("status") == "success":
        pon("pais", e.get("country"))
        pon("codigo", e.get("countryCode"))
        pon("region", e.get("regionName"))
        pon("ciudad", e.get("city"))
        pon("cp", e.get("zip"))
        pon("lat", e.get("lat"))
        pon("lon", e.get("lon"))
        pon("zona", e.get("timezone"))
        pon("isp", e.get("isp"))
        pon("org", e.get("org"))
        pon("asn", e.get("as"))
        pon("reverse", e.get("reverse"))
        pon("movil", e.get("mobile"))
        pon("proxy", e.get("proxy"))
        pon("hosting", e.get("hosting"))

    f = res.get("ipinfo") or {}
    pon("org", f.get("org"))
    pon("reverse", f.get("hostname"))
    pon("ciudad", f.get("city"))
    pon("region", f.get("region"))
    pon("pais", f.get("country"))
    pon("zona", f.get("timezone"))
    if f.get("loc") and not d.get("lat"):
        try:
            la, lo = f["loc"].split(",")
            pon("lat", float(la))
            pon("lon", float(lo))
        except Exception:
            pass

    g = res.get("freeip") or {}
    pon("pais", g.get("countryName"))
    pon("codigo", g.get("countryCode"))
    pon("ciudad", g.get("cityName"))
    pon("region", g.get("regionName"))
    pon("zona", g.get("timeZone"))
    if g.get("isProxy"):
        pon("proxy", True)

    il = res.get("iploc") or {}
    pon("pais", il.get("country_name"))
    pon("codigo", il.get("country_code2"))
    pon("isp", il.get("isp"))

    h = res.get("hostip") or {}
    pon("pais", h.get("country_name"))
    pon("codigo", h.get("country_code"))
    pon("ciudad", h.get("city"))
    return d


def _txt(nombre, tipo="TXT"):
    d = json.loads(urllib.request.urlopen(
        "https://dns.google/resolve?name=%s&type=%s"
        % (urllib.parse.quote(nombre), tipo), timeout=TIMEOUT).read().decode("utf-8", "replace")
    ) if True else {}
    return d


def _dns_json(nombre, tipo):
    try:
        url = ("https://dns.google/resolve?name=%s&type=%s"
               % (urllib.parse.quote(nombre), tipo))
        with urllib.request.urlopen(url, timeout=TIMEOUT) as r:
            return json.loads(r.read(200000).decode("utf-8", "replace"))
    except Exception:
        return {}


def _asn_cymru(ip):
    inv = ".".join(reversed(str(ip).split(".")))
    for reg in (_dns_json(inv + ".origin.asn.cymru.com", "TXT").get("Answer") or []):
        partes = [x.strip() for x in str(reg.get("data", "")).strip('"').split("|")]
        if len(partes) >= 5:
            return {"asn": partes[0], "prefijo": partes[1], "pais": partes[2],
                    "registro": partes[3], "fecha": partes[4]}
    return None


def _asn_nombre(asn):
    limpio = str(asn).replace("AS", "")
    for reg in (_dns_json("AS%s.asn.cymru.com" % limpio, "TXT").get("Answer") or []):
        partes = [x.strip() for x in str(reg.get("data", "")).strip('"').split("|")]
        if len(partes) >= 5:
            return {"asn": partes[0], "pais": partes[1], "registro": partes[2],
                    "nombre": partes[4]}
    return None


def cmd_ip(ctx, objetivo=None):
    ui = ctx.ui
    if not objetivo:
        ui.malo("uso: ip <ip|dominio>")
        return
    if es_ipv6(objetivo):
        ui.aviso("ipv6: cobertura limitada, se consulta igual")
    ip = resolver(objetivo)
    if not ip:
        ui.malo("no se pudo resolver: " + str(objetivo))
        ctx.emitir({"error": "no resuelto", "objetivo": str(objetivo)})
        return

    ui.titulo("IP INTEL · %s -> %s" % (objetivo, ip))
    datos = {"objetivo": str(objetivo), "ip": ip, "fecha": ahora()}

    if es_privada(ip):
        ui.aviso("direccion privada: sin geo ni reputacion")
        ui.campo("rango", "RFC1918 / loopback")
        ctx.archivar("ip", objetivo, datos)
        ctx.emitir(datos)
        return

    q = urllib.parse.quote(ip)
    res = ctx.anim.esperar("consultando 9 fuentes", lambda: paralelo({
        "geo": lambda: _geo(ctx, ip),
        "cymru": lambda: _asn_cymru(ip),
        "idb": lambda: ctx.red.json("https://internetdb.shodan.io/" + q),
        "otx": lambda: ctx.red.json(
            "https://otx.alienvault.com/api/v1/indicators/IPv4/%s/general" % q),
        "sfs": lambda: ctx.red.json("https://api.stopforumspam.org/api?ip=%s&json" % q),
        "scan": lambda: ctx.red.json("https://urlscan.io/api/v1/search/?q=ip:%s&size=20" % q),
        "rev": lambda: ctx.red.get("https://api.hackertarget.com/reverseiplookup/?q=" + q),
        "rdap": lambda: ctx.red.json("https://rdap.org/ip/" + q,
                                     headers={"Accept": "application/rdap+json"}),
        "ptr": lambda: _dns_json(ip + ".in-addr.arpa" if not ip.startswith("::") else ip, "PTR"),
    }), estilo="radar") or {}

    g = res.get("geo") or {}
    if not g and not res.get("cymru"):
        ui.malo("las fuentes no devolvieron datos")
        ctx.emitir({"error": "sin datos", "ip": ip})
        return

    cy = res.get("cymru")
    if cy:
        ui.sub("red")
        ui.campo("asn", "AS" + str(cy.get("asn")))
        ui.campo("prefijo", cy.get("prefijo"))
        ui.campo("registro rir", cy.get("registro"))
        ui.campo("pais del asn", cy.get("pais"))
        nom = _asn_nombre(cy.get("asn"))
        if nom:
            ui.campo("operador", nom.get("nombre"))
            datos["operador"] = nom.get("nombre")
        datos["asn"] = {"asn": cy.get("asn"), "prefijo": cy.get("prefijo"),
                        "rir": cy.get("registro")}
    if g.get("org"):
        ui.campo("org", g.get("org"))
    if g.get("isp"):
        ui.campo("isp", g.get("isp"))
    if g.get("empresa"):
        ui.campo("empresa", g.get("empresa"))

    if g.get("pais"):
        ui.sub("geolocalizacion")
        ui.campo("pais", "%s (%s)" % (g.get("pais"), g.get("codigo") or "??"))
        ui.campo("region", g.get("region"))
        ui.campo("ciudad", g.get("ciudad"))
        ui.campo("cp", g.get("cp"))
        if g.get("lat"):
            ui.campo("coord", "%s , %s" % (g.get("lat"), g.get("lon")))
            ui.mapa(g.get("lat"), g.get("lon"))
        if g.get("zona"):

def _comprobar(ctx, nombre, plantilla, tipo, aguja, usuario):
    url = plantilla.format(u=urllib.parse.quote(usuario))
    code, cuerpo = ctx.red.get(url, timeout=12)
    if code != 200:
        return nombre, url, False
    if tipo == "404":
        return nombre, url, True
    if tipo == "presente":
        return nombre, url, bool(aguja and aguja in cuerpo)
    if tipo == "ausente":
        return nombre, url, bool(aguja and aguja not in cuerpo)
    return nombre, url, True

    for ev in (d.get("events") or [])[:6]:
        ui.campo(ev.get("eventAction") or "evento", (ev.get("eventDate") or "")[:10])

    ui.sub("nameservers")
    for n in (d.get("nameservers") or [])[:6]:

def cmd_tls(ctx, dominio=None):
    ui = ctx.ui
    if not dominio:
        ui.malo("uso: tls dominio.com")
        return
    dominio = dominio_limpio(dominio)
    ui.titulo("TLS · " + dominio)

    def leer():
        ctx_s = ssl.create_default_context()
        ctx_s.check_hostname = False
        ctx_s.verify_mode = ssl.CERT_NONE
        with socket.create_connection((dominio, 443), timeout=10) as s:
            with ctx_s.wrap_socket(s, server_hostname=dominio) as t:
                return t.getpeercert()

    cert = ctx.anim.esperar("negociando tls", leer, estilo="ondas")
    if not cert:
        ui.malo("no se pudo obtener el certificado")
        return

    datos = {"dominio": dominio}
    ui.campo("emitido a", str(cert.get("subject")))
    ui.campo("emitido por", str(cert.get("issuer")))
    ui.campo("valido desde", cert.get("notBefore"))
    ui.campo("valido hasta", cert.get("notAfter"))
    datos["subject"] = str(cert.get("subject"))
    datos["issuer"] = str(cert.get("issuer"))
    datos["hasta"] = cert.get("notAfter")

    san = []
    for tipo, valor in cert.get("subjectAltName") or ():
        if tipo == "DNS":
            san.append(valor)
    if san:
        ui.sub("san (subdominios en el certificado)")
        for s in sorted(set(san))[:40]:
            ui.crudo("  " + Color.borde("├─ ") + Color.valor(s))
        datos["san"] = sorted(set(san))

    ui.sep()
    ui.enlace("crtsh", "https://crt.sh/?q=" + dominio)
    ctx.archivar("tls", dominio, datos)
    ctx.emitir(datos)

ALGOS = ("md5", "sha1", "sha256", "sha512", "sha3_256", "blake2b")


def cmd_hash(ctx, texto=None):
    ui = ctx.ui
    if not texto:
        ui.malo("uso: hash <texto|hash>")
        return
    ui.titulo("HASH · " + str(texto)[:40])
    datos = {"entrada": str(texto), "hashes": {}}
    limpio = str(texto).strip().lower()
    es_hash = bool(re.fullmatch(r"[0-9a-f]{8,128}", limpio))

    if not es_hash:
        for a in ALGOS:
            h = hashlib.new(a, str(texto).encode()).hexdigest()
            datos["hashes"][a] = h
            ui.campo(a, h)
    else:
        tipo = {32: "md5/ntlm", 40: "sha1", 64: "sha256", 96: "sha384",
                128: "sha512"}.get(len(limpio), "desconocido")
        ui.campo("identificado", tipo)
        datos["identificado"] = tipo
        ui.sub("lookups")
        ui.enlace("crackstation", "https://crackstation.net/")
        ui.enlace("md5decrypt", "https://md5decrypt.net/")
        ui.enlace("hashes.com", "https://hashes.com/en/decrypt/hash")
        ui.enlace("virustotal", "https://www.virustotal.com/gui/file/" + limpio)
        if len(limpio) in (32, 40):
            algo = "md5" if len(limpio) == 32 else "sha1"
            r = ctx.red.json("https://hashlookup.circl.lu/lookup/%s/%s" % (algo, limpio))
            if isinstance(r, dict) and r.get("FileName"):
                ui.campo("circl", r.get("FileName"))
                datos["circl"] = r.get("FileName")
            else:
                ui.info("circl sin coincidencia")
    ui.sep()
    ctx.archivar("hash", str(texto)[:40], datos)
    ctx.emitir(datos)

RE_CORREO = re.compile(r"[A-Za-z0-9._%+\-]+@[A-Za-z0-9.\-]+\.[A-Za-z]{2,}")
RE_TEL = re.compile(r"(?:\+\d{1,3}[\s.\-]?)?(?:\(?\d{2,4}\)?[\s.\-]?){2,5}\d{2,4}")
RE_REDES = {
    "instagram": re.compile(r"instagram\.com/([A-Za-z0-9_.]{2,40})"),
    "twitter": re.compile(r"(?:twitter|x)\.com/([A-Za-z0-9_]{2,20})"),
    "facebook": re.compile(r"facebook\.com/([A-Za-z0-9_.]{2,50})"),
    "linkedin": re.compile(r"linkedin\.com/(?:in|company)/([A-Za-z0-9_\-]{2,60})"),
    "github": re.compile(r"github\.com/([A-Za-z0-9_\-]{2,40})"),
    "tiktok": re.compile(r"tiktok\.com/@([A-Za-z0-9_.]{2,40})"),
    "telegram": re.compile(r"t\.me/([A-Za-z0-9_]{3,40})"),
    "youtube": re.compile(r"youtube\.com/(?:c/|channel/|@)([A-Za-z0-9_\-]{2,50})"),
    "whatsapp": re.compile(r"wa\.me/(\d{7,15})"),
}
RE_JS = re.compile(r'src=["\']([^"\']+\.js[^"\']*)["\']', re.I)
RE_COMENTARIO = re.compile(r"<!--(.*?)-->", re.S)
RE_META = re.compile(r'<meta[^>]+(?:name|property)=["\']([^"\']+)["\'][^>]+'
                     r'content=["\']([^"\']*)["\']', re.I)
RE_TITULO = re.compile(r"<title[^>]*>(.*?)</title>", re.S | re.I)
RE_GENERADOR = re.compile(r'<meta[^>]+name=["\']generator["\'][^>]+'
                          r'content=["\']([^"\']+)["\']', re.I)


def cmd_grab(ctx, url=None):
    ui = ctx.ui
    if not url:
        ui.malo("uso: grab https://sitio.com")
        return
    url = str(url)
    if not re.match(r"^https?://", url):
        url = "https://" + url
    host = urllib.parse.urlparse(url).netloc

    ui.titulo("GRAB · " + url)
    code, cuerpo = ctx.anim.esperar("descargando", lambda: ctx.red.get(url, cache=False),
                                    estilo="remolino")
    if code != 200 or not cuerpo:
        ui.malo("no se pudo descargar (codigo %s)" % code)
        return

    datos = {"url": url, "host": host, "codigo": code}
    titulo = RE_TITULO.search(cuerpo)
    ui.campo("titulo", titulo.group(1).strip()[:120] if titulo else "-")
    gen = RE_GENERADOR.search(cuerpo)
    if gen:
        ui.campo("generador", gen.group(1))

    ui.sub("metadatos")
    for nombre, valor in RE_META.findall(cuerpo)[:12]:
        ui.campo(nombre[:16], valor[:80])
        datos.setdefault("meta", {})[nombre] = valor

    correos = sorted(set(RE_CORREO.findall(cuerpo)))
    if correos:
        ui.sub("correos · %d" % len(correos))
        for c in correos[:20]:
            ui.campo("·", c)
        datos["correos"] = correos[:20]

    telefonos = sorted({t.strip() for t in RE_TEL.findall(cuerpo)
                        if len(re.sub(r"\D", "", t)) >= 8})[:15]
    if telefonos:
        ui.sub("telefonos")
        for t in telefonos:
            ui.campo("·", t)
        datos["telefonos"] = telefonos

    redes = {}
    for red, patron in RE_REDES.items():
        hallados = sorted(set(patron.findall(cuerpo)))
        if hallados:
            redes[red] = hallados[:8]
    if redes:
        ui.sub("perfiles sociales")
        for red, cuentas in redes.items():
            ui.campo(red, ", ".join(cuentas))
        datos["redes"] = redes

    js = sorted(set(RE_JS.findall(cuerpo)))[:15]
    if js:
        ui.sub("javascript")
        for j in js:
            ui.enlace("js", j[:100])
        datos["js"] = js

    comentarios = [c.strip()[:160] for c in RE_COMENTARIO.findall(cuerpo) if c.strip()][:10]
    if comentarios:
        ui.sub("comentarios html")
        for c in comentarios:
            ui.campo("·", c)
        datos["comentarios"] = comentarios

    ip = resolver(host)
    if ip:
        ui.campo("ip", ip)
        datos["ip"] = ip

    ui.sub("pivotes")
    ui.enlace("urlscan", "https://urlscan.io/domain/" + host)
    ui.enlace("wayback", "https://web.archive.org/web/*/" + url)
    ui.enlace("builtwith", "https://builtwith.com/" + host)
    ui.enlace("virustotal", "https://www.virustotal.com/gui/domain/" + host)
    ui.sep()
    ctx.archivar("grab", host, datos)
    ctx.emitir(datos)

def _pixel():
    return ('<img src="https://iplogger.org/1.png" width="1" height="1" '
            'alt="" style="display:block">')


def cmd_spam(ctx, destino=None):
    ui = ctx.ui
    cfg = ctx.config
    if not destino:
        ui.malo("uso: spam correo@destino.com [--asunto X] [--url Y]")
        return
    destino = str(destino).strip()
    if not correo_valido(destino):
        ui.malo("destino invalido")
        return

    host = cfg.get("spam_host")
    puerto = int(cfg.get("spam_puerto"))
    usuario = cfg.get("spam_usuario")
    clave = cfg.get("spam_clave")
    remitente = cfg.get("spam_de") or usuario

    if not usuario or not clave:
        ui.malo("configura primero: config spam_usuario / config spam_clave")
        return

    asunto = ctx.flags.get("asunto") or cfg.get("spam_asunto")
    url = ctx.flags.get("url") or "https://" + host
    cuerpo_txt = cfg.get("spam_cuerpo").replace("{url}", url)

    ui.titulo("SPAM · " + destino)
    ui.campo("servidor", "%s:%s" % (host, puerto))
    ui.campo("de", remitente)
    ui.campo("asunto", asunto)
    ui.campo("url", url)
    ui.sep()

    msg = EmailMessage()
    msg["From"] = remitente
    msg["To"] = destino
    msg["Subject"] = asunto
    msg["Message-ID"] = make_msgid()
    msg.set_content(cuerpo_txt)
    if cfg.get("spam_html", True):
        html = PLANTILLA_HTML.format(
            asunto=asunto, cuerpo=cuerpo_txt.replace("\n", "<br>"),
            url=url, pixel=_pixel() if cfg.get("spam_pixel", True) else "")
        msg.add_alternative(html, subtype="html")

    def enviar():
        try:
            if cfg.get("spam_ssl", False):
                s = smtplib.SMTP_SSL(host, puerto, timeout=25)
            else:
                s = smtplib.SMTP(host, puerto, timeout=25)
                if cfg.get("spam_inicio", True):
                    s.starttls(context=ssl.create_default_context())
            s.login(usuario, clave)
            s.send_message(msg)
            s.quit()
            return True
        except Exception as e:
            return str(e)

    r = ctx.anim.esperar("enviando", enviar, estilo="sobre")
    if r is True:
        ui.ok("enviado a " + destino)
        datos = {"destino": destino, "asunto": asunto, "url": url, "ok": True}
    else:
        ui.malo("fallo: %s" % r)
        datos = {"destino": destino, "asunto": asunto, "url": url, "error": str(r)}
    ui.sep()
    ctx.archivar("spam", destino, datos)
    ctx.emitir(datos)

FLAGS_CON_VALOR = {"asunto", "url"}


def partir(linea):
    partes = []
    actual = ""
    comillas = None
    for c in linea:
        if comillas:
            if c == comillas:
                comillas = None
            else:
                actual += c
        elif c in "\"'":
            comillas = c
        elif c.isspace():
            if actual:
                partes.append(actual)
                actual = ""
        else:
            actual += c
    if actual:
        partes.append(actual)
    return partes


def ejecutar(ctx, linea):
    partes = partir(linea)
    if not partes:
        return True
    objetivo = partes[0].lower()

    flags = {}
    resto = []
    i = 1
    while i < len(partes):
        p = partes[i]
        if p.startswith("--"):
            nombre = p[2:].lower()
            if nombre in FLAGS_CON_VALOR and i + 1 < len(partes):
                flags[nombre] = partes[i + 1]
                i += 2
                continue
            flags[nombre.replace("-", "_")] = True
            i += 1
            continue
        resto.append(p)
        i += 1
    ctx.flags = flags

    comando = ALIAS.get(objetivo)
    if not comando:
        ctx.ui.malo("comando desconocido: %s" % objetivo)
        ctx.ui.info("escribe ayuda")
        return True

    if comando == "salir":
        return False
    if comando == "limpiar":
        os.system("cls" if os.name == "nt" else "clear")
        cmd_banner(ctx)
        return True
    if comando == "banner":
        cmd_banner(ctx)
        return True

    argumento = " ".join(resto) if resto else None
    funcion = COMANDOS.get(comando)
    if not funcion:
        ctx.ui.malo("no implementado")
        return True
    try:
        funcion(ctx, argumento)
    except KeyboardInterrupt:
        ctx.ui.aviso("interrumpido")
    except Exception as e:
        ctx.ui.malo("error: %s" % e)
    return True

def main():
    os.makedirs(DIR_BASE, exist_ok=True)
    os.makedirs(RUTA_DOSSIERS, exist_ok=True)
    config = Config()
    ui = UI(config)
    red = Red(config)
    anim = Anim(config, ui)
    ctx = Contexto(config, ui, red, anim, {})

    argumentos = sys.argv[1:]
    if argumentos:
        ui.silencio = False
        cmd_banner(ctx)
        ejecutar(ctx, " ".join(argumentos))
        return

    cmd_banner(ctx)
    ui.info("escribe 'ayuda' para ver los comandos")
    while True:
        try:
            linea = input(Color.borde("jero") + Color.tenue(" › ") + " ").strip()
        except (EOFError, KeyboardInterrupt):
            ui.crudo()
            ui.info("hasta luego")
            break
        if not linea:
            continue
        guardar_historial(linea)
        if not ejecutar(ctx, linea):
            ui.info("hasta luego")
            break
